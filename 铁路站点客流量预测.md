def Deal_Fun(b):
    for m in range(b):

            if '始发日期' in data.iloc[i, 0]:
            	# '[0-9\—]+'正则表达式，匹配由0-9和长横线组成的一串连续字符
                # \—：一个特殊字符 —（长横线/破折号，不是普通的减号 -）
                # 加号 + 表示 前面的模式重复一次或多次
                # re.findall() 会返回一个 列表
                Head_d.append(re.findall('[0-9\—]+', data.iloc[i, 0]))
            if data.iloc[i, 0] == '上车站':
				#| `[A-Z]` | 匹配**一个大写英文字母**（A~Z）                     |
				#| `{2}`   | 表示前面的 `[A-Z]` **重复 2 次**，即匹配 **两个大写字母** |
				#| `[0-9]` | 匹配**一个数字字符**（0~9）                       |
				#| `{2}`   | 表示前面的 `[0-9]` **重复 2 次**，即匹配 **两个数字**   |
                Line.append(re.findall('[A-Z]{2}[0-9]{2}', data.iloc[i - 1, 0]))
        Line = pd.DataFrame(Line)  # 以数据框形式存放 车次
        Head_d = pd.DataFrame(Head_d) # 以数据框形式存放 开行日期
        Line['Head'] = 0  # 新加存放开行日期的列(Line.iloc[i,1];)，并且赋值为0
        # 通过循环赋值
        # 将开行日期第一行第一列的值赋给车次的第i行第2列
        for i in range(len(Line)):
            Line.iloc[i, 1] = Head_d.iloc[0, 0]
        # 提取列车站点信息
        # 提取"上车站"这个名称所在的行是哪一个
        on_station = [i for i, x in enumerate(data.iloc[:, 0]) if x == '上车站']
        # 提取"上车人数"这个名称所在的行是哪一个
        # on_count = []
		# for i, x in enumerate(data.iloc[:, 0]):
    	# 	  if x == '上车人数合计':
        #        on_count.append(i)
        on_count = [i for i, x in enumerate(data.iloc[:, 0]) if x == '上车人数合计']
        # 创建名称size的dataframe类型的数据框
        Size = pd.DataFrame(np.zeros([len(on_station), 2]), columns=['on_station', 'on_count'])
        Size['on_station'] = on_station
        Size['on_count'] = on_count
        # 利用DataFrame的「动态表格结构」机制多增加一列
        Size['off_count'] = 0
        # 执行完后，Size 会变成：

		# on_station	on_count    off_count
		#   	4          34           0
		#  		36	       59			0
		#   	61	       86			0
        # 对于每一组“上车站”所在的行，找到这一行中哪一列是“下车人数合计”，然后把那个列号（索引）填入 off_count。
        # [i for i, x in enumerate(data.iloc[Size.iloc[h, 0], :]) if x == '下车人数合计']
        # Size.iloc[h, 0]锁定上车站所在行；data.iloc[Size.iloc[h, 0], :]搜索所在行的所有列
        # 在某一行里，查找“下车人数合计”这一单元格所在的列号（索引）
        # [0]取第一个匹配的值
        for h in range(len(Size.iloc[:, 0])):
            Size.loc[h, 'off_count'] = [i for i, x in enumerate(
                data.iloc[Size.iloc[h, 0], :]) if x == '下车人数合计'][0]
         # 执行完后，Size 会变成：

		# on_station	on_count    off_count
		#    4            34           30
		#    36	          59		   22
		#    61	          86		   26
        #    ...           ...          ...
        
        # 下车站点
        # guodu 取数据框格式的原数据遍历的第一列
        # Size.iloc[j, 0]就是on_station列
        # Size.loc[j, 'on_count']就是on_count 列
        # guodu.iloc[Size.iloc[j, 0] + 2:Size.loc[j, 'on_count'], 0]
        guodu = pd.DataFrame(data.iloc[:, 0])
        off_station = []

            Station.loc[k + len(on_time[i]):k - 1 + len(off_time[i]), 'on_time'] = 0.1
            Station.loc[k + len(on_man[i]):k - 1 + len(off_man[i]), 'on_man'] = 0.1
            # 补充日期与车次信息
            Station.loc[k:k - 1 + len(off_man[i]), 'date'] = Line.iloc[i, 1]
            Station.loc[k:k - 1 + len(off_man[i]), 'train'] = Line.iloc[i, 0]
            k = k + len(off_man[i])
